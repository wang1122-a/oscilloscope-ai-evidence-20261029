cmake_minimum_required(VERSION 3.22)
project(english_asr LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(KALDI_NATIVE_FBANK_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(KALDI_NATIVE_FBANK_BUILD_PYTHON OFF CACHE BOOL "" FORCE)
set(SPM_ENABLE_SHARED OFF CACHE BOOL "" FORCE)
set(SPM_ENABLE_TCMALLOC OFF CACHE BOOL "" FORCE)
set(SPM_BUILD_TEST OFF CACHE BOOL "" FORCE)
add_subdirectory(vendor/kaldi-native-fbank EXCLUDE_FROM_ALL)
add_subdirectory(vendor/sentencepiece EXCLUDE_FROM_ALL)
add_library(rknnrt SHARED IMPORTED)
if(NOT RKNN_RUNTIME)
set(RKNN_RUNTIME "${CMAKE_CURRENT_SOURCE_DIR}/../../jniLibs/arm64-v8a/librknnrt.so")
endif()
set_target_properties(rknnrt PROPERTIES IMPORTED_LOCATION "${RKNN_RUNTIME}")
add_library(asr_core STATIC asr_core.cpp word_lm.cpp)
target_include_directories(asr_core PUBLIC . vendor/kaldi-native-fbank vendor/sentencepiece/src)
target_link_libraries(asr_core PUBLIC kaldi-native-fbank-core sentencepiece-static rknnrt log)
add_library(english_asr SHARED asr_jni.cpp)
target_link_libraries(english_asr PRIVATE asr_core)
add_executable(asr_probe asr_probe.cpp)
target_link_libraries(asr_probe PRIVATE asr_core)
