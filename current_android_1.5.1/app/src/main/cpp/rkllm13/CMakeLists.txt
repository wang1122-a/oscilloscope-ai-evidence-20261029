cmake_minimum_required(VERSION 3.22.1)
project(xiaou_rkllm13 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(PREBUILT "${CMAKE_CURRENT_SOURCE_DIR}/../../jniLibs/${ANDROID_ABI}")
add_library(rkllmrt SHARED IMPORTED)
set_target_properties(rkllmrt PROPERTIES IMPORTED_LOCATION "${PREBUILT}/librkllmrt.so")
add_library(omp_private SHARED IMPORTED)
set_target_properties(omp_private PROPERTIES IMPORTED_LOCATION "${PREBUILT}/libomp.so")
add_library(rkllm_bridge SHARED rkllm_bridge.cpp)
set(SV_STL "${CMAKE_ANDROID_NDK}/sources/cxx-stl/llvm-libc++/libs/${CMAKE_ANDROID_ARCH_ABI}")
target_link_libraries(rkllm_bridge PRIVATE -Wl,--whole-archive "${SV_STL}/libc++_static.a" "${SV_STL}/libc++abi.a" -Wl,--no-whole-archive rkllmrt omp_private log dl m)
target_compile_options(rkllm_bridge PRIVATE -O3 -fno-fast-math -fvisibility=hidden -fvisibility-inlines-hidden)
target_link_options(rkllm_bridge PRIVATE -Wl,--no-undefined -Wl,--exclude-libs,ALL -Wl,-Bsymbolic "-Wl,--version-script=${CMAKE_CURRENT_SOURCE_DIR}/rkllm_exports.map")
